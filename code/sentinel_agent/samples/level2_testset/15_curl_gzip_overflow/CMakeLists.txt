cmake_minimum_required(VERSION 3.10)
project(gziprelay VERSION 8.11.1 LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
include_directories(include)
add_library(gziprelay_lib STATIC src/gziprelay.c)
add_executable(gziprelay src/main.c)
target_link_libraries(gziprelay gziprelay_lib)
enable_testing()
add_test(NAME smoke COMMAND gziprelay)
